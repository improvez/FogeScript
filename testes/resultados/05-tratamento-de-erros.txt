// O arquivo completo é testes/resultados/05-tratamento-de-erros.js: este programa mais o runtime (funções foge.*).

'use strict';
// Programa FogeScript traduzido para JavaScript (tradutor 1.2.0).
// Contas com tipos garantidos pela análise semântica usam os operadores do JavaScript.
// As outras chamam o runtime (foge.*), que confere os tipos na hora de rodar.
// O último número de cada chamada foge.* é a linha do código FogeScript.

// Funções (NÃO, EU NUNCA FIZ ISSO)
function dividir(a, b) {
  if (b === 0) {
    throw foge.lancar("não divido por zero", 7);
  }
  return foge.conta("/", a, b, 9);
}
foge.funcao("dividir", dividir);

// A SENHORA TRABALHA NA ASSEMBLEIA?
function assembleia() {
  let a;
  let b;
  let lista = [10, 20, 30];
  for (let vez = 1; vez <= 3; vez++) {
    foge.passo(15);
    a = foge.ler(16);
    b = foge.ler(16);
    try {
      foge.gravar(foge.formatar(a), "/", foge.formatar(b), "=", foge.chamar("dividir", [a, b], 18));
    } catch ($erro) {
      let motivo = foge.capturar($erro);
      foge.gravar("A senhora tá nervosa:", motivo);
    }
  }
  try {
    foge.gravar(foge.item(lista, b, 26));
  } catch ($erro) {
    let motivo = foge.capturar($erro);
    foge.gravar("Erro capturado:", motivo);
  }
  foge.gravar("Fim da sessão.");
}
assembleia();

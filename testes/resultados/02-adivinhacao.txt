// O arquivo completo é testes/resultados/02-adivinhacao.js: este programa mais o runtime (funções foge.*).

'use strict';
// Programa FogeScript traduzido para JavaScript (tradutor 1.2.0).
// Contas com tipos garantidos pela análise semântica usam os operadores do JavaScript.
// As outras chamam o runtime (foge.*), que confere os tipos na hora de rodar.
// O último número de cada chamada foge.* é a linha do código FogeScript.

// Variáveis globais
let DIAS_TRABALHADOS = 3;
let MAX_TENTATIVAS = 5;

// A SENHORA TRABALHA NA ASSEMBLEIA?
function assembleia() {
  let palpite;
  let tentativas = 0;
  let acertou = false;
  foge.gravar("Repórter: Quantos dias a senhora trabalhou este mês?");
  while (!acertou && tentativas < MAX_TENTATIVAS) {
    foge.passo(16);
    palpite = foge.ler(17);
    tentativas++;
    if (palpite === DIAS_TRABALHADOS) {
      acertou = true;
    } else if (foge.conta(">", palpite, DIAS_TRABALHADOS, 21)) {
      foge.gravar(palpite, "dias? Menos, senhora, menos.");
    } else {
      foge.gravar(palpite, "dias? Mais que isso!");
    }
  }
  if (acertou) {
    foge.gravar("TEMOS GRAVADO! Foram", DIAS_TRABALHADOS, "dias. Acertou em", tentativas, "tentativa(s).");
  } else {
    foge.gravar("Ela fugiu antes de responder.");
  }
}
assembleia();

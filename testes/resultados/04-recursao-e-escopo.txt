function fibonacci(n) {
  chamadas++;
  if (foge.conta("<", n, 2, 8)) {
    return n;
  }
  return foge.conta("+", foge.chamar("fibonacci", [foge.conta("-", n, 1, 11)], 11), foge.chamar("fibonacci", [foge.conta("-", n, 2, 11)], 11), 11);
}
foge.funcao("fibonacci", fibonacci);

function mdc(a, b) {
  while (b !== 0) {
    foge.passo(15);
    let resto = foge.conta("%", a, b, 16);
    a = b;
    b = resto;
  }
  return a;
}
foge.funcao("mdc", mdc);

'use strict';
// Programa FogeScript traduzido para JavaScript (tradutor 1.2.0).
// Contas com tipos garantidos pela análise semântica usam os operadores do JavaScript.
// As outras chamam o runtime (foge.*), que confere os tipos na hora de rodar.
// O último número de cada chamada foge.* é a linha do código FogeScript.

// Funções (NÃO, EU NUNCA FIZ ISSO)
function somar(lista) {
  let total = 0;
  for (let i = 0; i < foge.chamar("TAMANHO", [lista], 8); i++) {
    foge.passo(8);
    total = foge.acumular("+=", foge.item(lista, i, 9), total, 9);
  }
  return total;
}
foge.funcao("somar", somar);

function classificar(media) {
  if (foge.conta(">=", media, 8, 15)) {
    return "funcionária exemplar";
  } else if (foge.conta(">=", media, 4, 17)) {
    return "meio expediente";
  }
  return "funcionária fantasma";
}
foge.funcao("classificar", classificar);

// A SENHORA TRABALHA NA ASSEMBLEIA?
function assembleia() {
  let nome;
  let dias;
  let horas = [];
  nome = foge.ler(27);
  dias = foge.ler(27);
  for (let d = 1; foge.conta("<=", d, dias, 28); d++) {
    foge.passo(28);
    let h;
    h = foge.ler(30);
    foge.guardar(h, foge.lista(horas, 31), foge.chamar("TAMANHO", [horas], 31), 31);
  }
  let total = foge.chamar("somar", [horas], 34);
  let media = foge.conta("/", total, foge.chamar("TAMANHO", [horas], 35), 35);
  foge.gravar("Funcionária:", nome);
  foge.gravar("Horas por dia:", horas);
  foge.gravar("Total:", total, "horas | Média:", media);
  foge.gravar("Veredito:", foge.chamar("classificar", [media], 39));
}
assembleia();
